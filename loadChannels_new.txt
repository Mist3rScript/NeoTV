
    loadChannels: async (page = 1) => {
        app.curPage = page;
        try {
            const response = await fetch(`${app.api}/admin/channels?page=1&limit=10000`);
            const data = await response.json();
            const container = document.getElementById('channels-folder-view');
            
            if (!container) {
                console.error('Container channels-folder-view not found');
                return;
            }
            
            if (!data.data || data.data.length === 0) {
                container.innerHTML = '<div style="text-align:center; padding:40px; color:#666;"><i class="fas fa-folder-open" style="font-size:3rem; display:block; margin-bottom:15px; color:#444;"></i>No channels yet. Add from Catalog!</div>';
                return;
            }
            
            // Group by category
            const grouped = {};
            data.data.forEach(ch => {
                const cat = ch.category || 'Uncategorized';
                if (!grouped[cat]) grouped[cat] = [];
                grouped[cat].push(ch);
            });

            // Render folders
            const html = [];
            Object.keys(grouped).sort().forEach(category => {
                const channels = grouped[category];
                const folderId = 'folder-' + category.replace(/\s+/g, '-').replace(/[^a-zA-Z0-9-]/g, '');
                
                html.push('<div>');
                html.push(`<div class="folder-item" onclick="app.toggleFolder('${folderId}')">`);
                html.push('<i class="fas fa-folder folder-icon"></i>');
                html.push(`<span class="folder-name">${category}</span>`);
                html.push(`<span class="folder-count">${channels.length} channel${channels.length > 1 ? 's' : ''}</span>`);
                html.push('<i class="fas fa-chevron-right folder-toggle"></i>');
                html.push('</div>');
                html.push(`<div class="folder-channels" id="${folderId}">`);
                
                channels.forEach(c => {
                    const logo = c.logo_url 
                        ? `<img src="${c.logo_url}">` 
                        : '<i class="fas fa-tv" style="font-size:1.2rem; color:#444;"></i>';
                    
                    html.push('<div class="channel-item">');
                    html.push(`<div class="channel-logo">${logo}</div>`);
                    html.push('<div class="channel-info">');
                    html.push(`<div class="channel-name">${c.name}</div>`);
                    html.push(`<div class="channel-url">${c.stream_url || ''}</div>`);
                    html.push('</div>');
                    html.push('<div class="channel-actions">');
                    html.push(`<button class="btn-stream" onclick='app.openEditChannel(${JSON.stringify(c).replace(/'/g, "&#39;")})'><i class="fas fa-edit"></i></button>`);
                    html.push(`<button class="btn-stream danger" onclick="app.deleteChannel(${c.id})"><i class="fas fa-trash"></i></button>`);
                    html.push('</div>');
                    html.push('</div>');
                });
                
                html.push('</div>');
                html.push('</div>');
            });
            
            container.innerHTML = html.join('');
            console.log('Folders rendered:', Object.keys(grouped).length);
            
        } catch (e) { 
            console.error('Load channels error:', e);
        }
    },

    toggleFolder: (folderId) => {
        const folderEl = document.getElementById(folderId);
        if (folderEl) {
            const folderItem = folderEl.previousElementSibling;
            folderEl.classList.toggle('open');
            folderItem.classList.toggle('open');
        }
    },
